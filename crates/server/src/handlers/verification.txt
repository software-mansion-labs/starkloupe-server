fn read_json(file_path: PathBuf) -> anyhow::Result<Value> {
    let sierra_file = File::open(file_path).context("Unable to open json file")?;

    serde_json::from_reader(sierra_file).context("Unable to read json file")
}

pub async fn verify_handler(
    chain_id: Path<String>,
    class_hash: Path<String>,
) -> Result<Json<Option<u8>>, StatusCode> {
    // TODO: get Sierra bytecode from Node

    let provider_client = create_rpc_client(chain_id.to_string());

    let a = provider_client.get_class(
        BlockId::Number(777),
        FieldElement::from_str(&class_hash.to_string().as_str()).unwrap(),
    ).await;

    // TODO: compile Cairo project to Sierra bytecode

    let path_to_scarb_project = "tmp/starknet_multiple_contracts";

    let mut cmd = ScarbCommand::new_with_stdio();
    cmd.current_dir(path_to_scarb_project);
    cmd.arg("build");
    let a = cmd.run();
    dbg!(&a);

    // TODO: compare the two bytecodes

    // TODO: add Sierra Contract Class to the database

    let sierra_json = read_json(PathBuf::from("tmp/starknet_multiple_contracts/target/dev/starknet_multiple_contracts_Balance.contract_class.json"))
        .unwrap();
    let sierra_class: ContractClass = serde_json::from_value(sierra_json).unwrap();
    dbg!(&sierra_class);
    let sierra_program = sierra_class.extract_sierra_program().unwrap();
    dbg!(&sierra_program);

    let program = sierra_compiler::compile(sierra_class, usize::MAX);

    dbg!(program.debug_info.sierra_statement_info);

    Ok(Json(None))
}
